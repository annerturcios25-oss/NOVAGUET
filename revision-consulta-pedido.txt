function generarHashContrasena(contrasena) {
  return new Promise((resolve, reject) => {
    const salt = crypto.randomBytes(16).toString("hex");

    crypto.scrypt(contrasena, salt, 64, (error, clave) => {
      if (error) {
        return reject(error);
      }

      resolve(salt + ":" + clave.toString("hex"));
    });
  });
}
